rgbSpeed = 3
rgbHouseEnabled = false

r34.AddSlider(r34, {
    ["Name"] = "Velocidade do RGB",
    ["Min"] = 1,
    ["Max"] = 26,
    ["Default"] = 3,
    ["Callback"] = function(value, ...)
        rgbSpeed = value
        return
    end
})

local function getHsvColor(speed, ...)
    return Color3.fromHSV(tick() * speed % 10 / 10, 1, 1)
end

local function fireRemote(remoteName, args, ...)
    local remoteFolder = replicatedStorage:FindFirstChild("RE")

    if remoteFolder then
        local remote = remoteFolder:FindFirstChild(remoteName)

        if remote then
            pcall(function(...)
                remote:FireServer(unpack(args))
                return
            end)
        end
    end

    return
end

r34.AddToggle(r34, {
    ["Name"] = "RGB na Casa",
    ["Default"] = false,
    ["Callback"] = function(state, ...)
        rgbHouseEnabled = state

        if state then
            task.spawn(function(...)
                while rgbHouseEnabled do
                    fireRemote("1Player1sHous1e", {
                        "ColorPickHouse",
                        getHsvColor(rgbSpeed)
                    })
                    task.wait(0.1)
                end
                return
            end)
        end

        return
    end
})
